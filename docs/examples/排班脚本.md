# 排班脚本

按开始时间生成志愿者岗位列表。空行分开输入和输出，较长的说明可以折行显示。

```javascript
const shifts = [
  { start: "09:30", desk: "门厅", team: "布置" },
  { start: "10:00", desk: "前台", team: "接待" },
  { start: "10:30", desk: "阅读角", team: "朗读" },
];

const formatShift = ({ start, desk, team }) =>
  `${start}  ·  ${desk}  ·  ${team}`;

const reminder = "换班前五分钟请到门厅长桌核对物料数量、岗位卡和未完成的交接事项，确认每位志愿者都知道下一班的联系人。";

console.log(shifts.map(formatShift).join("\n"));
console.log(reminder);
```
